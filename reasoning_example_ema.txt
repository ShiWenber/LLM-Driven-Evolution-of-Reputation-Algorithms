def evaluate(current_reputation, observation, my_history, round_num):
    alpha = 0.9  # weight on existing reputation; higher = slower change
    if observation['action'] == 'cooperate':
        score = 1.0
    else:
        score = -1.0
    return alpha * current_reputation + (1 - alpha) * score

def decide(recipient_reputation, round_num, my_history):
    # Count how many times we have cooperated in the past
    cooperations = sum(1 for entry in my_history if entry['action'] == 'cooperate')
    total_past = len(my_history)
    # Our own cooperation rate, smoothed to avoid division by zero
    coop_rate = cooperations / (total_past + 1)
    # Threshold becomes lower (more forgiving) the more we cooperate ourselves
    threshold = 0.2 * (1 - coop_rate)
    return recipient_reputation > threshold
