    """
    Update reputation using context-aware scores:
    - Donating to a low-reputation recipient is more positive.
    - Not donating to a high-reputation recipient is more negative.
    Learning rate decays with rounds to stabilize.
    """
    # Decaying learning rate: start high, slowly decrease
    alpha = 0.5 * (0.9 ** (round_num // 50))  # half-life roughly 50 rounds
    alpha = max(0.05, min(0.5, alpha))        # clamp to [0.05, 0.5]

    # Default scores if no recipient reputation available
    donate_score = 1.0
    not_donate_score = -1.0

    # Adjust scores based on recipient's reputation (if available)
    recipient_rep = observation.get('recipient_reputation', 0.0)
    if observation['action'] == 'donate':
        # More positive if recipient reputation is low (generous act)
        donate_score = 1.0 + 0.5 * max(-1.0, -recipient_rep)
        new_rep = (1 - alpha) * current_reputation + alpha * donate_score
    else:
        # More negative if recipient reputation is high (stingy)
        not_donate_score = -1.0 - 0.5 * max(0.0, recipient_rep)
        new_rep = (1 - alpha) * current_reputation + alpha * not_donate_score

    # Small symmetric noise for flexibility
    noise = random.uniform(-0.03, 0.03)
    new_rep += noise
    return max(-1.0, min(1.0, new_rep))
